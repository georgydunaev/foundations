\subsection{Definitions}
Context: (it may be ignored potentially, but it is an ``intended use'')\\
The parameter $A$ is an arbitrary class(so no assumptions, just a note that it is not necessary a set, it may be a proper class),\\
The parameter $R$ is an arbitrary class and a relation, i.e. $R\subseteq V\times V$.
\begin{definition}[irreflexive on]
\label{def:irreflexiveOn}
\[
(R\mathrm{\ Irr\ }A)\iffdef (\forall x\in A: (x,x)\notin R)
\]
\end{definition}
\begin{definition}[relation is transitive on]
\label{def:transitiveOn}
\[
(R\mathrm{\ Trans\ }A)\iffdef (\forall x\in A\forall y\in A\forall z\in A:
(x,y)\in R \land (y,z)\in R \implies (x,z)\in R)
\]
\end{definition}

\begin{definition}[relation is strictly partially ordered on]
\label{def:partiallyOrderedOn}
\[
(R\mathrm{\ Po\ }A)\iffdef ((R\mathrm{\ Irr\ }A) \land (R\mathrm{\ Trans\ }A)))
\]
\end{definition}
\begin{definition}[relation is linear on]
\label{def:linearOn}
\[
(R\mathrm{\ Lin\ }A)\iffdef (\forall x\in A\forall y\in A:
((x,y)\in R) \lor (x=y) \lor ((y,x)\in R))
\]
\end{definition}
\begin{definition}[relation is strictly completely ordered on]
\label{def:sOn}
\[
(R\mathrm{\ Or\ }A)\iffdef ((R\mathrm{\ Po\ }A) \land (R\mathrm{\ Lin\ }A)))
\]
\end{definition}
\begin{definition}[relation is founded on]
\label{def:foundedOn}
\[
(R\mathrm{\ Fr\ }A)
\iffdef
(\forall S\subseteq A:
S\subseteq R[S]
\implies
S=\emptyset)
\]
\end{definition}
\begin{definition}[relation is well-ordered on]
\label{def:wellOrderedOn}
\[
(R\mathrm{\ We\ }A)
\iffdef
((R\mathrm{\ Or\ }A) \land (R\mathrm{\ Fr\ }A)))
\]
\end{definition}

\begin{definition}[class is transitive]
\label{def:transitiveClass}
\[
\Tr(X)
\iffdef
(X\subseteq \power X)
\]
\end{definition}

\begin{definition}[epsilon relation]
\label{def:epsilonRelation}
\[
\E\eqdef \{(x,y)\ |\ x\in y\}
\]
\end{definition}
\begin{definition}[class is ordinal]
\label{def:ordinalClass}
\[
\Ord(X)
\iffdef
((\Tr(X))\land(\E\mathrm{\ We\ }X))
\]
\end{definition}
\begin{definition}[limit class]
Note, that this definition is a bit unconventional: usually it must also be nonempty.
\label{def:limitClass}
\[
\LimClass(X)
\iffdef
\left(X\subseteq\union X\right)
\]
\end{definition}
\begin{definition}[limit ordinal]
Note, that this definition is a bit unconventional: usually it must also be nonempty.
\label{def:limitClass}
\[
\LimOrd(X)
\iffdef
((\Ord(X))\land(\LimClass(X))
\]
\end{definition}

\begin{definition}[ordinal numbers] It can be interchangebly called ``ordinal sets''.
\label{def:ordinalNumbers}
\[
\On\ \eqdef\ \{\,x\ |\ \Ord(x)\,\}
\]
\end{definition}

\section{Ordinals and induction principles}
This section organizes the ordinal constructions around an arbitrary
relation, then specializes to membership. The aim is to make the assumptions
behind each induction principle visible. The notation distinguishes a
transitive relation from a class closed under taking predecessors, and an
ordinal class from an ordinal set. The elementary comparison results come
first; the later recursion, collapse, and cardinal results state their
additional set-existence and choice assumptions explicitly. They are not
consequences of ordinal notation alone.
\subsection{Ordinal classes in general setting}
Let $A$ be a class, possibly proper, and let
$R\subseteq\V\times\V$ be a relation on sets. Write $xRy$ for
$(x,y)\in R$. Relational images use the orientation
$R[S]=\cc{y}{\exists x\in S,\ xRy}$, so $R^\top[\{y\}]$ is the class of
predecessors of $y$. The experimental family-based notation immediately
below is an alternative representation; subsequent relation proofs use
the ordinary pair-membership interpretation unless explicitly stated.
\begin{definition}[related]
\[
ARB \iffdef (A,B) \in R
\labeltagd{def}{related}
\]
\end{definition}
\begin{definition*}[related (alternative family representation)]
\[
ARB \iffdef \exists (i\set),\ (R((i,0))=A)\land(R((i,1))=B)
\labeltagd{def}{relatedexper}
\]
\end{definition*}
\begin{definition}[epsilon relation]
\[
\E\ \eqdef\ \cc{(x,y)}{(y\set)\land(x\in y)}
\labeltagd{def}{epsilonRelation}
\]
\end{definition}
\begin{definition}[irreflexive on]
\[
(R\mathrm{\ Irr\ }A)\iffdef (\forall x\in A,\ \neg (xRx))
\labeltagd{def}{irreflexiveOn}
\]
\end{definition}
\begin{definition}[relation is transitive on]
\[
(R\mathrm{\ Trans\ }A)\iffdef (\forall x\in A\forall y\in A\forall z\in A,\ %
xRy \land yRz \implies xRz)
\labeltagd{def}{transitiveOn}
\]
\end{definition}

\begin{definition}[relation is strictly partially ordered on]
\[
(R\mathrm{\ Po\ }A)\iffdef ((R\mathrm{\ Irr\ }A) \land (R\mathrm{\ Trans\ }A))
\labeltagd{def}{partiallyOrderedOn}
\]
\end{definition}
\begin{definition}[relation is linear on]
\[
(R\mathrm{\ Lin\ }A)\iffdef (\forall x\in A\forall y\in A:
(xRy) \lor (x=y) \lor (yRx))
\labeltagd{def}{linearOn}
\]
\end{definition}
\begin{definition}[relation is strictly linearly ordered on]
\[
(R\mathrm{\ Or\ }A)\iffdef ((R\mathrm{\ Po\ }A) \land (R\mathrm{\ Lin\ }A))
\labeltagd{def}{sOn}
\]
\end{definition}
\begin{definition}[relation is founded on]
The condition says that every nonempty subclass of $A$ has an element with
no predecessor in that subclass.
\[
(R\mathrm{\ Fr\ }A)
\iffdef
(\forall S\subclass A,\ %
S\subclass R[S]
\implies
S=\emptyclass)
\labeltagd{def}{foundedOn}
\]
\end{definition}
\begin{definition}[relation is well--ordered on]
\[
(R\mathrm{\ We\ }A)
\iffdef
((R\mathrm{\ Or\ }A) \land (R\mathrm{\ Fr\ }A))
\labeltagd{def}{wellOrderedOn}
\]
\end{definition}



\begin{definition}[$R_1R_2$ subclass]
\[
A\stackrel{R_1R_2}{\subclass} B \iffdef  (\forall x,\ xR_1A\implies xR_2B)
\labeltagd{def}{RRsubclass}
\]
\end{definition}
\begin{definition}[$R$ subclass]
\[
(A\stackrel{R}{\subclass} B)
\iffdef
(\forall x,\ xRA\implies x\in B)
\labeltagd{def}{Rsubclass}
\]
\end{definition}
The target $B$ may be a proper class. Therefore the final membership
test is primitive membership, not $x\E B$: the relation $\E$ only
contains records whose second coordinate is a set. When $B$ is a set,
this definition agrees with $A\stackrel{R\E}{\subclass}B$; for a
proper $B$ those expressions need not agree. For example, a nonempty
predecessor class is contained in $\V$, although none of its elements
is $\E$-related to the proper class $\V$.
\begin{lemma}[$R$ subclass]
\[
k\stackrel{R}{\subclass} X \iff  R^{\top}[\{k\}]\subclass X
\labeltagd{lem}{Rsubset}
\]
\end{lemma}
\begin{proof}
Unfolding the definition, both sides say that every $R$-predecessor of $k$
belongs to $X$.
\end{proof}
\begin{definition}[powerclass w.r.t. relation $R$]
\[
\power_R X\ \eqdef\ \cc{ k}{k\stackrel{R}{\subclass}X}
\labeltagd{def}{RpowerClass}
\]
\end{definition}
\begin{lemma}[Membership predecessors and the powerclass]
For every class $X$, including proper classes,
\[
\power_{\E}X=\power X.
\]
\end{lemma}
\begin{proof}
Both comprehensions contain only sets. For a set $k$, the defining
condition for $k\in\power_{\E}X$ is that every $x$ with $x\E k$
belongs to $X$. Since $k$ is a set, $x\E k$ is equivalent to
$x\in k$. The condition is therefore exactly $k\subseteq X$,
which defines $k\in\power X$. Class extensionality proves the equality.
\end{proof}

\begin{definition}[class is closed under R-predecessors]
This is the meaning of ``$R$-transitive class'' here. It is different from
transitivity of $R$ as a binary relation.
\[
(R\ \Tr\ K)
\iffdef
(K\subclass \power_R K)
\labeltagd{def}{RtransitiveClass}
\]
\end{definition}

\[
(

\land
(B\subclass A)

(R\mathrm{\ Irr\ }B)
\labeltagd{thm}{IrrInheritable}
\]\[

(R\mathrm{\ Trans\ }A)
\land
(B\subclass A)

\implies
(R\mathrm{\ Trans\ }B)
\labeltagd{thm}{TransInheritable}
\]
\[
(
(R\mathrm{\ Po\ }A)
\land
(B\subclass A)
)
\implies
(R\mathrm{\ Po\ }B)
\labeltagd{thm}{PoInheritable}
\]
\[
(
(R\mathrm{\ Fr\ }A)
\land
(B\subclass A)
)
\implies
(R\mathrm{\ Fr\ }B)
\labeltagd{thm}{FrInheritable}
\]
\[
(
(R\mathrm{\ We\ }A)
\land
(B\subclass A)
)
\implies
(R\mathrm{\ We\ }B)
\labeltagd{thm}{WeInheritable}

\end{theorem}
\begin{proof}

a subclass can omit a predecessor that was present in $A$.
\end{proof}

\begin{theorem}[induction on a R--transitive and R--well--founded class]
\[
(R\ \Tr\ K)
\land
(R\mathrm{\ Fr\ }K)
\implies
(
((K\cap(\power_R A))\subclass A)
\implies
(K\subclass A)
)
\labeltagd{thm}{Rind}
\]
\end{theorem}
\begin{proof}
Assume $K$ is closed under $R$-predecessors, $R$ is founded on $K$, and
$K\cap\power_R A\subseteq A$. If $K\nsubseteq A$, the nonempty class
$S=K\setminus A$ has an $R$-minimal element $m$: $m\notin R[S]$.
Since $m\in K$, all predecessors of $m$ lie in $K$. None lies in $S$,
so all lie in $A$. Thus $m\in\power_R A$, and the induction assumption
implies $m\in A$, contradicting $m\in S$.
\end{proof} %the proof is finished

\begin{corollary}[alternative induction principle on $R_1$--well--founded class]
This formulation of induction corresponds to Exercise 9.22 of ``Introduction to Axiomatic Set Theory'' by Takeuti and Zaring.
\[
(R_1\mathrm{\ Fr\ } K)
\implies
(
((K\cap(\power_{(R_1\mathrm{\ on\ }K)} A))\subclass A)
\implies
(K\subclass A)
)
\labeltagd{thm}{RindAlt}
\]
\begin{proof}
Set $R=R_1\mathrm{\ on\ }K$. The preceding lemma makes $K$
$R$-predecessor-closed. Since $R\subseteq R_1$, any nonempty subclass of
$K$ with an $R_1$-minimal element also has an $R$-minimal element. Thus
$R_1$-foundedness implies $R$-foundedness. Apply
\myref{thm:Rind} to $R$ and expand its definition.
\end{proof}
\end{corollary}

\subsection{Specialization to membership}
Now take $R=\E$. The predicate $\Ord(X)$ applies to classes, while
$\On$ collects only the set instances of that predicate. The limit
convention deliberately includes the empty ordinal. The operations
$\sup$ and $\inf$ below first discard nonordinal members of their argument;
in particular, $\sup(\emptyclass)=\emptyclass$ and
$\inf(\emptyclass)=\On$.

\begin{definition}[class is transitive (w.r.t. epsilon relation)]
\[
\Tr(X)
\iffdef
(X\subclass \power X)
\labeltagd{def}{transitiveClass}
\]
\end{definition}


\begin{definition}[class is ordinal (w.r.t. epsilon relation)]
\[
\Ord(X)
\iffdef
((\Tr(X))\land(\E\mathrm{\ We\ }X))
\labeltagd{def}{ordinalClass}
\]
\end{definition}
\begin{definition}[limit class]
This convention includes the empty class; the usual limit convention also requires nonemptiness.
\[
\LimClass(X)
\iffdef
\left(X\subclass\union X\right)
\labeltagd{def}{limitClass}
\]
\end{definition}
\begin{definition}[limit ordinal]
This convention includes zero; the usual definition of a limit ordinal also requires nonemptiness.
\[
\LimOrd(X)
\iffdef
((\Ord(X))\land(\LimClass(X)))
\labeltagd{def}{limitOrdinal}
\]
\end{definition}

\begin{definition}[ordinal numbers] These are also called ``ordinal sets''.
\[
\On\ \eqdef\ \cc{x}{\Ord(x)}
\labeltagd{def}{ordinalNumbers}
\]
\end{definition}
\paragraph{Comprehension for ordinal sets.}
The expanded predicate $\Ord(x)$ contains a class quantifier through
foundedness, so its literal use in comprehension is permitted by MK.
Under heredity, the displayed definition also has an elementary
form suitable for NBG comprehension. Indeed, for a set $x$, every
class $S\subseteq x$ is a set. Foundedness on $x$ is therefore
equivalent to saying that every nonempty \emph{set} $S\subseteq x$
has a membership-minimal element. The remaining transitivity and
ordering conditions quantify only over set elements. Replacing the
foundedness clause by this equivalent set-quantified condition constructs
$\On$ by elementary comprehension. This reduction concerns the set
instances collected in $\On$; it does not restrict the predicate
$\Ord$ itself to sets or assume global regularity.

\begin{definition}[supremum]
\[
\sup(A)\ \eqdef\ \bigcup\left(\On\cap A\right)
\labeltagd{def}{supremum}
\]
\end{definition}
\begin{definition}[infimum]
\[
\inf(A)\ \eqdef\ \On\cap\left(\bigcap (\On\cap A)\right)
\labeltagd{def}{infimum}
\]
\end{definition}

\begin{theorem}[infimum of a subset of ordinals is the unique minimum]
\[
((A\cap\On)\neq\emptyclass)\implies(\inf(A)\in A)
\labeltagd{thm}{infordmin}
\]
\begin{proof}
Use the well-ordering of $\On$ under membership, recorded below as
\myref{thm:ordon}. The nonempty class $C=A\cap\On$ has a least member
$m$. For every $q\in C$, ordinal comparison gives $m\subseteq q$.
Hence $m\subseteq\bigcap C$. Since $m\in C$, also
$\bigcap C\subseteq m$. Therefore
$\inf(A)=\On\cap\bigcap C=\On\cap m=m\in A$.
This argument makes the ordinal well-ordering prerequisite explicit.
\end{proof}
\end{theorem}

global regularity is not silently introduced.

\begin{lemma}[epsilon relation criterion]
\[
((x\in y)\land(y\set)) \iff ((x,y)\in\E)
\labeltagd{lem}{ordlem0}
\]
\end{lemma}
\begin{proof}
If $x\in y$ and $y$ is a set, then $x$ is a set as well; the set pair
$(x,y)$ belongs to the comprehension defining $\E$. Conversely, membership
in that relation supplies set components $u,v$ with $u\in v$ and
$(x,y)=(u,v)$. Injectivity of the chosen set-pair coding gives $x=u$ and
$y=v$, hence $x\in y$ and $y$ is a set.
\end{proof}

\begin{lemma}[an element is a proper subset]
\[
((\E\mathrm{\ Irr\ }A)\land(\Tr(A)))
\implies
(
(B\in A)
\implies
(B\subsetneq A))
\labeltagd{lem}{ordlem1}
\]
\end{lemma}
\begin{proof}
Let $B\in A$. Transitivity gives $B\subseteq A$. If $B=A$, then
$A\in A$, so $A$ is itself an element of the carrier $A$ and
irreflexivity on that carrier gives $A\notin A$. Thus $B\ne A$.
\end{proof}
\begin{lemma}[a proper subclass is an element]
%(\mathrm{ })\land
\[
(
(\Tr(A))\land
(\E\mathrm{\ Fr\ }A)\land
(\E\mathrm{\ Lin\ }A)\land
(\Tr(B))
)
\implies
(
(B\subsetneq A)
\implies
(B\in A)
)
\clabeltagd{lem}{ordlem2}
\]
\end{lemma}
\begin{proof}
Let $B\subsetneq A$. The nonempty class $A\setminus B$ has an
$\E$-minimal member $m$. Since $m\in A$ and $A$ is transitive,
$m\subseteq A$. Minimality excludes every member of $m$ from
$A\setminus B$, hence $m\subseteq B$.

For the reverse inclusion, take $b\in B$. Both $b$ and $m$ belong to
$A$, so linearity gives $b\in m$, $b=m$, or $m\in b$. The second
alternative puts $m$ in $B$; the third does so by transitivity of $B$.
Both contradict $m\notin B$. Therefore $b\in m$, so $B\subseteq m$.
Extensionality yields $B=m\in A$.
\end{proof}

\begin{theorem}[proper subclass iff element]
\[
(
(\Ord(A))\land
(\Tr(B))
)
\implies
(
(B\in A)
\iffnows
(B\subsetneq A)
)
\labeltagd{thm}{subiffel}
\]
\end{theorem}
\begin{proof}
An ordinal class is transitive, and membership on it is irreflexive,
linear, and founded. Apply the preceding two lemmas.
\end{proof}
\begin{theorem}[transitive subclass of ordinal numbers iff it is an ordinal class]
\[
(
\Ord(A)
)
\iff
(
(A\subclass\On)
\land
(\Tr(A))
)
\clabeltagd{thm}{dford5}
\]
\url{https://us.metamath.org/mpeuni/dford5.html}
\end{theorem}
\begin{proof}
If $\Ord(A)$, then every member of $A$ is a set ordinal by
\myref{thm:tz7.6}; therefore $A\subseteq\On$, and transitivity is part
of the definition. Conversely, if $A\subseteq\On$ and $A$ is transitive,
its membership order is inherited from the well-order of $\On$, proved
in \myref{thm:ordon}. Hence $\Ord(A)$. The proof of that latter theorem
uses the preceding proper-subclass criterion, not this characterization.
\end{proof}
\begin{theorem}[element of ordinal is ordinal]
\[
\Ord(A)\land(a\in A)\implies \Ord(a)
\clabeltagd{thm}{tz7.6}
\]%(7.6 Takeuti--Zaring)
\end{theorem}
\begin{proof}
Let $a\in A$ and $\Ord(A)$. Transitivity of $A$ gives $a\subseteq A$,
so membership inherits its strict well-order from $A$. To check
transitivity of $a$, take $z\in y\in a$. All three objects $z,y,a$
lie in $A$, by transitivity of $A$. Transitivity of membership on $A$
then gives $z\in a$. Thus $a$ is transitive and well-ordered by membership.
\end{proof}
\begin{theorem}[class of ordinal numbers is ordinal class]
\[
(
\Ord(\On)
)
\labeltagd{thm}{ordon}
\]
\end{theorem}
\begin{proof}
First compare any two ordinal classes $A,B$. Their intersection
$C=A\cap B$ is transitive. If $C$ were a proper subclass of both,
\myref{thm:subiffel} would give $C\in A$ and $C\in B$, hence $C\in C$.
Since $C\in A$, this contradicts irreflexivity on $A$. Thus $A\subseteq B$
or $B\subseteq A$; if the inclusion is proper, the smaller class is a
member of the larger, again by \myref{thm:subiffel}.

If $a\in\On$ and $b\in a$, then $b$ is a set and is ordinal by
\myref{thm:tz7.6}, so $b\in\On$. Hence $\On$ is transitive.
The comparison just proved gives linearity of membership on $\On$;
transitivity follows because its members are transitive. A self-member
$a\in a$ would contradict irreflexivity on the ordinal carrier $a$.

Finally, let $S\subseteq\On$ be nonempty and choose $a\in S$.
If $S\cap a$ is empty, then $a$ is minimal in $S$. Otherwise choose
its least member $m$ using the well-order on $a$. Any $b\in S\cap m$
would also lie in $a$, by transitivity of $a$, contradicting minimality
of $m$ in $S\cap a$. Thus $m$ is minimal in $S$. Membership on $\On$
is founded, completing the proof of $\Ord(\On)$. No global regularity
axiom was needed: foundedness was included in the definition of each
ordinal class.
\end{proof}

\begin{theorem}[criterion in the context of the regularity axiom]
Under global regularity, ordinality has the following shorter characterization.
\[
(\E\mathrm{\ Fr\ }V)
\implies
\left(
\Ord(X) \iff \Tr(X)\land (\forall Y\in X: \Tr(Y))
\right)
\labeltagd{thm}{ordViaReg}
\]
\end{theorem}
\begin{proof}
The forward direction follows from transitivity and the fact that each
member of an ordinal class is ordinal, hence transitive.

Conversely, suppose $X$ and every member of $X$ are transitive.
Regularity makes membership founded on $X$. If
$a\in X\cap\power\On$, then $a$ is transitive and $a\subseteq\On$;
\myref{thm:dford5} gives $\Ord(a)$. Since $a$ is a set, $a\in\On$.
Thus $X\cap\power\On\subseteq\On$. Apply the induction theorem
\myref{thm:Rind} to the transitive, founded class $X$, with target class
$\On$. It yields $X\subseteq\On$, and \myref{thm:dford5} now gives
$\Ord(X)$.
\end{proof}

\section{Cardinals}
Here cardinality is extended from sets to classes by assigning $\On$ when
no ordinal set receives an injection from the argument. This is a local
convention for a class-valued functional symbol; $\On$ is not being declared
a set cardinal. The intended set-cardinal behavior requires the stated
well-ordering or size assumptions. Comparison of all classes additionally
uses the relevant global choice or limitation-of-size principle.

\subsection{Axioms}
\begin{iaxiom}[set cardinality is an ordinal set]
\[
|\cdot|:\V \to \On
\]
Since $|\cdot|$ is a functional symbol, the intended assertion is:
\[
(x\mapsto |x|):\V \to \On
\]
The functional symbol accepts any class as its argument, while
$(x\mapsto |x|)$ is defined on sets only. Under the stated set-size
assumption, this class function has graph
\[
\cc{\langle x, |x|\rangle }{ x \any\set}.
\]
\end{iaxiom}
Equivalently, we can state the axiom pointwise:
\begin{axiom*}[alternative set-size formulation]
\[
(x\set)\implies(|x|\in\On)
\]
\end{axiom*}
We can also require only sethood of the value:
\begin{axiom*}[alternative set-size formulation]
\[
(x\set)\implies(|x|\set)
\]
\end{axiom*}
The following theorem explains why this last formulation suffices.
\begin{theorem*}[cardinality is an ordinal]
\[
(x\set)\implies \Ord(|x|)
\]
\end{theorem*}
\begin{proof}
Put $C=\cc{a\in\On}{x\preccurlyeq a}$. If $C$ is nonempty, the
infimum theorem identifies $|x|=\inf C$ with a member of $C$, hence an
ordinal set. If $C$ is empty, the definition gives $|x|=\On$, which is
an ordinal class by \myref{thm:ordon}. Thus ordinality holds in either
case; the additional set-size axiom excludes the second case for set
arguments.
\end{proof}
\begin{iaxiom}[heredity]
Sets are hereditary:
\[
(A\subseteq B)\land (B\set) \implies (A\set)
\]
\end{iaxiom}

\subsection{Implementation}
The following definitions implement the proposed cardinality symbol.
\begin{definition}[cardinality of a class]
\[
|K|\eqdef \On \cap \bigcap \cc{a\in\On}{K\preccurlyeq a}
\]
or even
\[
|K|\eqdef \inf(\cc{a\in\On}{K\preccurlyeq a})
\]
in the sense of definition \ref{def:infimum}.
\end{definition}
\begin{definition}[relation ``not stronger in cardinality'']
Let $X$ and $Y$ be terms, then
\[
(X\preccurlyeq Y)
\iff
(\exists f,\ (f:X\to Y)\land (f \injective))
\labeltagd{def}{notstrongercard}
\]
\end{definition}
\begin{proposition}[Elementary comprehension for ordinal-set bounds]
Assume heredity, the set-image principle, and ordinary set-pair
formation. For every class $K$ and set $a$, the assertion
$K\preccurlyeq a$ is equivalent to the existence of a \emph{set}
graph that is an injection from $K$ into $a$. Consequently the primary
candidate class $\cc{a\in\On}{K\preccurlyeq a}$ exists by elementary
NBG comprehension under these assumptions. No choice principle is needed.
\end{proposition}
\begin{proof}
Let $f:K\to a$ be a class injection. Its range $D\subseteq a$ is a
set by heredity. The inverse graph is a class function on $D$, with
image $K$, so set-image makes $K$ a set. Ordinary set-pair formation
and elementary comprehension give the class function
$x\mapsto(x,f(x))$ on $K$. Its image is exactly the graph $f$;
set-image therefore makes $f$ a set. The converse is immediate, since
a set graph is also a class graph.

\begin{theorem}[Schroeder--Bernstein for classes]
Assume MK comprehension and let $f:A\to B$ and $g:B\to A$ be
injective class functions. Then there is a class bijection $h:A\to B$.
\end{theorem}
\begin{proof}
Put $A_0=A\setminus g[B]$. Let $C$ be the least subclass of $A$
containing $A_0$ and closed under $g\circ f$. It exists by the
intersection construction for a monotone operation, and its fixed-point
identity is
\[
C=A_0\cup g[f[C]].
\]
For completeness, intersect all subclasses $Q$ of $A$ with
$A_0\subseteq Q$ and $g[f[Q]]\subseteq Q$. The intersection has these
properties; applying minimality to $A_0\cup g[f[C]]$ proves the identity.
Define $h(a)=f(a)$ for $a\in C$ and $h(a)=g^{-1}(a)$ for
$a\in A\setminus C$. The latter value exists uniquely because
$A_0\subseteq C$ and $g$ is injective.

The two branches are injective. Their images are disjoint: if
$f(c)=g^{-1}(a)$ with $c\in C$ and $a\notin C$, then
$a=g(f(c))\in C$, a contradiction. Thus $h$ is injective.
For surjectivity, take $b\in B$. If $b\in f[C]$, it is already in
$h[C]$. Otherwise $g(b)\notin C$: if it belonged to $C$, it could not
belong to $A_0$, so the displayed identity would give
$g(b)=g(f(c))$ for some $c\in C$, contradicting injectivity of $g$
and $b\notin f[C]$. Hence $h(g(b))=b$ on the second branch.
This proves surjectivity. All graphs use set coordinates; no proper
class is collected as a member of another class.
\end{proof}

\end{proof}

\subsection{Induction principles}
\begin{note}[successor--inductive (Peano--inductive) classes and $\omega$ as the $\subseteq$--minimal successor--inductive class]
Because several distinct induction principles appear in class theory (countable Peano successor--induction, transfinite $\in$--induction on ordinals, and $\power$--induction on well--founded sets), the predicate $\mathrm{Ind}(X)$ (Entry 3.55 in the Obelisk catalog) uses ``inductive'' in its narrowest possible sense, namely \textbf{successor--inductive} (or \textbf{Peano--inductive} / $0$--successor--closed):
\[
\mathrm{Ind}(X) \iffdef (\emptyclass \in X) \land (\forall (a:\Class),\ (a \in X) \implies ((a \cup \{a\}) \in X)).
\]
By its definition $\forall (x:\Class),\ (x \in \omega) \iff (\forall (Q:\Class),\ \mathrm{Ind}(Q) \implies (x \in Q))$ (Entry 3.56) together with $\mathrm{Ind}(\omega)$ (Entry 3.59), $\omega$ is not only the smallest successor--inductive ordinal, but also the \textbf{$\subseteq$--minimal (least) element among all successor--inductive classes in the subclass ($\subseteq$) order}:
\[
\mathrm{Ind}(\omega) \land \forall (Q:\Class),\ \mathrm{Ind}(Q) \implies (\omega \subseteq Q),
\]
which directly expresses simple countable mathematical (Peano) induction up to $\omega$.
\end{note}
\begin{proposition}[The least successor-inductive class and Infinity]
\label{thm:omega_set_limit}
Assume MK comprehension, empty set, pairing, union, and heredity.
The class
\[
\omega=\{x:\forall Q,\ \mathrm{Ind}(Q)\implies x\in Q\}
\]
is successor-inductive and is contained in every successor-inductive
class. It is a transitive class of ordinal sets. If additionally
Infinity supplies a successor-inductive set, then $\omega$ is a set
ordinal and is a nonzero limit ordinal. Conversely, sethood of this
$\omega$ supplies the set required by Infinity.
\end{proposition}
\begin{proof}
Every successor-inductive class contains zero, so $0\in\omega$.
If $a\in\omega$, every such class contains $a$ and therefore its
successor $a\cup\{a\}$. Pairing and union make that successor a set,
so it belongs to $\omega$ by comprehension. The definition also gives
the asserted minimality.

Zero is an ordinal, and the successor of an ordinal set is an ordinal
set: it is a transitive set of ordinal sets, so the ordinal
characterization applies. Thus $\On$ is successor-inductive and
minimality gives $\omega\subseteq\On$. For transitivity put
\[
T=\{a\in\omega:a\subseteq\omega\}.
\]
Zero belongs to $T$. If $a\in T$, then its successor belongs to
$\omega$ and is a subclass of $\omega$, because both $a\subseteq\omega$
and $a\in\omega$ hold. Thus $T$ is successor-inductive. Minimality
gives $\omega\subseteq T$, proving transitivity of $\omega$.

If $I$ is a successor-inductive set, minimality gives $\omega\subseteq I$,
and heredity makes $\omega$ a set. Its transitivity and containment
in $\On$ make it an ordinal set. It is nonempty because it contains
zero. For each $a\in\omega$, its successor belongs to $\omega$ and
contains $a$, so $a\in\bigcup\omega$. Transitivity gives the reverse
inclusion, hence $\omega=\bigcup\omega$ is a nonzero limit ordinal.
Finally, if $\omega$ is a set, its already proved successor-inductiveness
is itself a witness to Infinity.
\end{proof}
\begin{proposition}[From set regularity to class foundedness]
\label{thm:set_regularity_class_foundedness}
Assume Infinity, pairing, union, heredity, set-image, class comprehension,
and the set-valued ordinal recursion theorem in the induction and
recursion chapter. Then ordinary set
regularity, namely that every nonempty set has a membership-minimal
element, is equivalent to foundedness of membership on $\V$: every
nonempty class has a membership-minimal element.
\end{proposition}
\begin{proof}
Only the forward implication needs proof. Let $S$ be a nonempty class
and choose $a\in S$. We first build a transitive set containing $a$
as an element, without assuming that $a$ already belongs to $\W$.
By recursion on the set $\omega$, form
\[
B_0=\{a\},\qquad B_{n+1}=\bigcup B_n,\qquad
T=\bigcup\{B_n:n\in\omega\}.
\]
Pairing and union make each $B_n$ a set. Set-image on $\omega$ and
union make $T$ a set, with $a\in T$. If $z\in T$, then
$z\in B_n$ for some $n$, and every member of $z$ belongs to
$B_{n+1}\subseteq T$. Thus $T$ is transitive. This construction uses
only ordinal recursion and the stated set-formation axioms, not
membership induction or the regularity conclusion being established.

The class $D=S\cap T$ is a nonempty set by heredity, since $a\in D$.
Set regularity supplies $m\in D$ with $m\cap D=\emptyclass$.
If $z\in m\cap S$, then $m\in T$ and transitivity give $z\in T$,
so $z\in m\cap D$, a contradiction. Hence $m\cap S=\emptyclass$,
as required. Conversely, apply class foundedness to each nonempty set.
\end{proof}
This bridge permits a proof stated with set regularity to use a minimal
element of a proper class only after the additional set-closure
prerequisites have been supplied. The ordinal comparison proofs earlier
in this chapter instead use foundedness built into their ordinal
definition; they do not need this global regularity principle.

\subsection{Recursion on a set-like founded relation}
The following generalization uses the class comprehension, Infinity,
pairing, union, heredity, set-image, and ordinal-recursion principles used
in the induction and recursion section. Global regularity is not needed:
foundedness is imposed on the particular relation being used. Auxiliary
graphs use ordinary Kuratowski pairs.

\begin{definition}[Set-like relation]
For $R\subseteq A\times A$ and $a\in A$, put
\[
P_a=\cc{b\in A}{bRa}.
\]
The relation is set-like on $A$ when each $P_a$ is a set. A subclass $D$
of $A$ is predecessor-closed when $a\in D$ implies $P_a\subseteq D$.
\end{definition}

\begin{theorem}[Set-valued recursion on a founded class]
\label{thm:setlike_class_recursion}
Let $R$ be founded and set-like on $A$, and let
$H:A\times\V\to\V$ be a class function. There is a unique class
function $F:A\to\V$ such that
\[
F(a)=H(a,F\restriction P_a)\qquad(a\in A).
\]
\end{theorem}
\begin{proof}
First form the least predecessor-closed set $T_a$ containing $a$.
Starting with $B_0=\{a\}$, put
\[
B_{n+1}=\bigcup\{P_b:b\in B_n\},\qquad
T_a=\bigcup\{B_n:n<\omega\}.
\]
Recursion on $\omega$ constructs this sequence independently of recursion
on $R$. Set-likeness and set-image make the family of predecessor sets
at each step a set; union and set-image then make $T_a$ a set.
It is predecessor-closed because the predecessors of a member of $B_n$
belong to $B_{n+1}$. Induction on $n$ proves minimality.

An approximation is a set function $p$ on a predecessor-closed set
$D\subseteq A$ satisfying $p(t)=H(t,p\restriction P_t)$ for $t\in D$.
Any two approximations agree on their common domain. Otherwise an
$R$-minimal disagreement has all its predecessors in both domains, with
equal values there, and hence equal values after applying $H$, a
contradiction.

Founded induction now constructs an approximation on $T_a$. Suppose the
unique approximations $p_b$ on $T_b$ exist for all $b\in P_a$.
Set-image on the set $P_a$ collects these graphs, and their union $q$ is
a set function by compatibility. Its domain
$D=\bigcup_{b\in P_a}T_b$ is predecessor-closed and contains $P_a$;
compatibility also preserves all its recursion equations. If $a\in D$,
retain $q$. Otherwise adjoin $(a,H(a,q\restriction P_a))$. The domain
remains predecessor-closed, and old equations are unaffected because
no member of $D$ has $a$ as a predecessor. Restrict to $T_a$, which is
contained in this domain by minimality. This gives the desired
approximation. The induction used here follows from a minimal
counterexample, or equivalently from \myref{thm:RindAlt}.

Collecting all approximation pairs by class comprehension produces $F$.
Compatibility makes it a function, and the solutions on $T_a$ give its
domain and equation. Each restriction to $P_a$ is part of such a set
approximation. Finally, a minimal disagreement between two class solutions
is impossible by the same argument, proving uniqueness.
\end{proof}

\subsection{Mostowski collapse}
\begin{theorem}[Collapse of a set-like extensional founded relation]
\label{thm:mostowski_collapse}
Let $R\subseteq A\times A$ be founded and set-like. Assume it is
extensional on $A$: $P_a=P_b$ implies $a=b$ for $a,b\in A$.
There is a unique transitive class $M$ and a unique isomorphism
\[
\pi:(A,R)\longrightarrow(M,\in).
\]
If $A$ is a set, then $M$ is a set.
\end{theorem}
\begin{proof}
Apply \ref{thm:setlike_class_recursion} with
$H(a,g)=\operatorname{ran}(g)$ on set function graphs, extending it by
the empty set on other inputs. Set-image makes these values sets.
The result satisfies
\[
\pi(a)=\{\pi(b):bRa\}.
\]
We prove injectivity by founded induction on $a$, with the assertion
quantified over \emph{every} $c\in A$: if $\pi(a)=\pi(c)$, then $a=c$.
For $bRa$, equality of the displayed sets gives $dRc$ with
$\pi(b)=\pi(d)$. The induction hypothesis at $b$ gives $b=d$, so
$bRc$. Conversely, for $dRc$ equality gives some $bRa$ with the same
image, and the same induction hypothesis gives $d=b$. Hence
$P_a=P_c$, and extensionality gives $a=c$. No induction hypothesis at
$c$ has been assumed.

Let $M=\pi[A]$. The recursion equation makes $M$ transitive, and
injectivity gives both directions of
$bRa\iff\pi(b)\in\pi(a)$. Thus $\pi$ is the required isomorphism.
If another isomorphism $\rho$ has transitive range, every member of
$\rho(a)$ belongs to that range and so is $\rho(b)$ for some $b\in A$.
The isomorphism property then gives
$\rho(a)=\{\rho(b):bRa\}$. Recursion uniqueness yields $\rho=\pi$,
and their ranges are equal. When $A$ is a set, set-image makes $M$ a set.
\end{proof}

\begin{example}[Why the class collapse needs set-likeness]
Order a copy of $\On$ and adjoin one new greatest point. More explicitly,
use the carrier $(\{0\}\times\On)\cup\{(1,0)\}$, order its first
part by the ordinal coordinate, and put every member of that part below
$(1,0)$. This is a founded extensional class order, but the greatest
point has a proper class of predecessors. In a transitive collapse the
point $(0,\alpha)$ would have image $\alpha$, by ordinal induction.
The greatest point would therefore have image $\On$, which is proper
and cannot be the value of a class function into $\V$. Thus dropping
set-likeness would make the stated collapse theorem false.
\end{example}

\begin{corollary}[Ordinal type of a set well-order]
\label{thm:set_wellorder_type}
Every strictly well-ordered set $(X,R)$ is isomorphic to a unique ordinal
set. Its isomorphism to that ordinal is unique.
\end{corollary}
\begin{proof}
The predecessor sets are subclasses of the set $X$. They distinguish
points: if $xRy$, then $x\in P_y$ but $x\notin P_x$.
Thus the collapse theorem applies. Its image is a transitive set whose
membership relation is a well-order, hence an ordinal set by definition.
Any ordinal isomorphism is a collapse to a transitive range, so uniqueness
also follows from that theorem.
\end{proof}

\begin{lemma}[Increasing maps do not shorten ordinals]
If $e:\alpha\to\beta$ is strictly increasing between ordinal sets, then
$\xi\leq e(\xi)$ for every $\xi<\alpha$, and $\alpha\leq\beta$.
Consequently the order type of a subset of $\beta$ is at most $\beta$.
\end{lemma}
\begin{proof}
By ordinal induction, for each $\eta<\xi$ we have
$\eta\leq e(\eta)<e(\xi)$. Thus every member of $\xi$ belongs to
$e(\xi)$, proving $\xi\subseteq e(\xi)$. If $\beta<\alpha$, applying
this at $\xi=\beta$ contradicts $e(\beta)<\beta$.
The increasing enumeration of a subset exists by its ordinal order type,
so the same conclusion applies to it.
\end{proof}

\begin{theorem}[The least ordinal bound is an ordinal representative]
Let $X$ be a set that injects into some ordinal set. For the primary
ordinal-set definition, $\kappa=|X|$ is equinumerous with $X$ and is
an initial ordinal: no smaller ordinal is equinumerous with $\kappa$.
Moreover $|\kappa|=\kappa$.
\end{theorem}
\begin{proof}
The nonempty class of candidate ordinal bounds has least member
$\kappa$, so there is an injection $f:X\to\kappa$. Give its image
$S$ the inherited well-order. Its ordinal type $\beta$ is at most
$\kappa$ by the lemma. The isomorphism $S\to\beta$ composed with $f$
is a bijection $X\to\beta$. Hence $\beta$ is a candidate bound and
minimality gives $\kappa\leq\beta$. Thus $\beta=\kappa$ and the
required bijection exists.

If an ordinal $\gamma<\kappa$ received an injection from $\kappa$,
composition with this bijection would make $\gamma$ a candidate for
$X$, a contradiction. In particular no smaller ordinal is equinumerous
with $\kappa$. The identity makes $\kappa$ its own candidate bound,
and the preceding exclusion of smaller bounds gives $|\kappa|=\kappa$.
\end{proof}

\begin{corollary}[Comparison of well-orderable sets]
For well-orderable sets $X,Y$,
\[
X\preccurlyeq Y\iff |X|\leq|Y|.
\]
They are equinumerous exactly when $|X|=|Y|$.
\end{corollary}
\begin{proof}
Transport an injection along the bijections to their ordinal representatives.
If $|Y|<|X|$, such an injection contradicts the minimality of $|X|$.
Conversely, ordinal inclusion gives an injection whenever
$|X|\leq|Y|$. Equal representatives give a bijection by composition;
a bijection gives injections in both directions and hence equality.
\end{proof}

\begin{remark}[The comparison convention needs its hypotheses]
The earlier notation $X\prec Y$ was defined by
$\neg(Y\preccurlyeq X)$. Under comparability this is equivalent to
$X\preccurlyeq Y$ together with the failure of the reverse injection.
Without comparability, two incomparable classes would satisfy both
$X\prec Y$ and $Y\prec X$. Thus the notation is not an unconditional
strict-order assertion in a setting without the stated size principles.
For arbitrary classes one may instead explicitly use
$X\preccurlyeq Y\land\neg(Y\preccurlyeq X)$ when that asymmetric
meaning is intended.
\end{remark}

\begin{proposition}[When the two cardinality proposals agree]
\label{prop:cardinality_proposals_agree}
Write $c(K)$ for the primary ordinal-set definition and $d(K)$ for the
predicate-indexed intersection over ordinal classes. If
$K\preccurlyeq\On$, then $c(K)=d(K)$. If no injection
$K\to\On$ exists, then
\[
c(K)=\On,\qquad d(K)=\V.
\]
In particular, an assumption giving $\V\preccurlyeq\On$ makes the two
proposals agree for all classes.
\end{proposition}
\begin{proof}
Every ordinal class is a subclass of $\On$, since its members are
ordinal sets. If such a class $C$ is a proper subclass of $\On$, choose
$\alpha\in\On\setminus C$. Ordinal comparison gives $C\subseteq\alpha$
(the alternative $\alpha\subsetneq C$ would give $\alpha\in C$).
Heredity makes $C$ a set. Thus the ordinal classes are precisely the
ordinal sets and $\On$.

If there is an ordinal-set bound for $K$, its least member belongs to
both collections of bounds and is contained in all their members.
Both intersections therefore give that least member. If the only
ordinal-class bound is $\On$, then $d(K)=\On$ and the empty-candidate
default also gives $c(K)=\On$. Finally, if $K$ does not inject into
$\On$, it cannot inject into any ordinal class; the predicate-indexed
intersection is then the empty intersection $\V$, whereas the primary
formula explicitly intersects its empty-intersection value with $\On$.
Every class embeds in $\V$ by inclusion, proving the last assertion.
\end{proof}

\subsection{Hartogs' bound and the set well-ordering theorem}
This subsection additionally assumes Powerset. The bound is constructed
without choice; a choice function enters only in the subsequent enumeration.

\begin{theorem}[An ordinal too large to inject into a given set]
\label{thm:hartogs_bound}
For every set $X$ there is a least ordinal $\mathfrak h(X)$ which does
not inject into $X$.
\end{theorem}
\begin{proof}
All pairs $(B,R)$ with $B\subseteq X$ and $R\subseteq X\times X$
form a set, by powerset and the ordinary set-product construction.
Separate those with $R\subseteq B\times B$ for which $R$ is a strict
well-order on $B$. Their unique
ordinal types form a set $H$ by \ref{thm:set_wellorder_type} and
set-image. An ordinal belongs to $H$ exactly when it injects into $X$:
an order isomorphism supplies such an injection, and any injection
transports the ordinal order onto its image in $X$.

If $\alpha\in H$ and $\beta\in\alpha$, restrict the injection
$\alpha\to X$ to $\beta$ to get $\beta\in H$. Thus $H$ is a
transitive set of ordinals, hence itself an ordinal. It cannot inject
into $X$, since that would give $H\in H$, contradicting ordinal
irreflexivity. Every smaller ordinal is a member of $H$ and does inject
into $X$. Set $\mathfrak h(X)=H$.
\end{proof}

\begin{theorem}[Well-ordering from a choice function]
If a set $X$ has a choice function $c$ on its nonempty subsets, then
$X$ can be well-ordered. Conversely, a well-order of $X$ supplies such
a choice function.
\end{theorem}
\begin{proof}
Use transfinite recursion to select
\[
f(\alpha)=
\begin{cases}
c(X\setminus\operatorname{ran}(f\restriction\alpha)),
 & X\setminus\operatorname{ran}(f\restriction\alpha)\ne\emptyclass,\\
\emptyclass,&\text{otherwise}.
\end{cases}
\]
The step is a set-valued rule on earlier set graphs, so the recursion
theorem applies. If the remainder were nonempty at every
$\alpha<\mathfrak h(X)$, these choices would be distinct members of
$X$, giving an injection $\mathfrak h(X)\to X$, a contradiction.
Let $\gamma<\mathfrak h(X)$ be the least stage with empty remainder.
Before that stage the choices are distinct, and exhaustion says that
they cover $X$. Hence $f\restriction\gamma$ is a bijection
$\gamma\to X$. Transport the ordinal well-order along this bijection.
The case $X=\emptyclass$ has $\gamma=0$.

Conversely, send each nonempty subset of a well-ordered $X$ to its least
element. Uniqueness and set-image give a choice function on the set
$\power X\setminus\{\emptyclass\}$.
\end{proof}

\begin{corollary}[The strength of the set-size assumption]
Under these set-formation assumptions, the following statements are
equivalent: every set can be well-ordered; every set injects into an
ordinal set; and every set $X$ has $|X|\in\On$ for the ordinal-set
cardinality convention of this chapter. They are also equivalent to the
axiom of choice for set-indexed families of nonempty sets.
\end{corollary}
\begin{proof}
A set well-order gives an ordinal isomorphism; an injection into an ordinal
pulls back a well-order. In the cardinality definition, the candidate
class is nonempty exactly when such an injection exists. Its infimum is
then its least member, an ordinal set; if it is empty, the value is the
proper class $\On$. Properness follows because $\Ord(\On)$ together
with sethood would give $\On\in\On$.

Choice applied to all nonempty subsets of $X$ gives the choice function
used above. Conversely, for a set-indexed family of nonempty sets,
well-order their union and choose the least element of each member.
The union is a set, and set-image collects the choices into a function.
This proves the equivalence with set choice, without introducing a
global choice function on proper classes.
\end{proof}

\subsection{Ordinal addition and the height convention}
\label{sec:ordinal_addition}
We use the set-valued transfinite-recursion theorem from the induction
and recursion chapter, together with pairing, union, and set-image. That
theorem does not use ordinal addition. The following construction therefore
also supplies the arithmetic facts used later in that chapter's height
discussion, despite their present placement in the book.

The stages used in this recursion need no prior arithmetic. The successor
$\alpha\cup\{\alpha\}$ is a transitive set of ordinal sets, hence an
ordinal set. The union of a set of ordinals is likewise an ordinal set:
it is transitive and contained in $\On$, and union supplies sethood.
Every nonzero ordinal $\delta$ is either a successor or a nonzero limit.
Indeed, if $\delta\ne\bigcup\delta$, choose
$\eta\in\delta\setminus\bigcup\delta$. Ordinal comparison forces
every member of $\delta$ to be at most $\eta$, since $\eta\in\xi\in\delta$
would contradict the choice of $\eta$. Thus
$\delta=\eta\cup\{\eta\}$, with unique predecessor $\eta$.
Otherwise $\delta=\bigcup\delta$ is the limit case.

\begin{theorem}[Ordinal addition by recursion]
For each ordinal set $\alpha$, there is a unique class function
$\beta\mapsto\alpha+\beta$ from $\On$ to $\On$ satisfying
\[
\begin{aligned}
\alpha+0&=\alpha,\\
\alpha+(\beta+1)&=(\alpha+\beta)\cup\{\alpha+\beta\},\\
\alpha+\lambda&=\bigcup_{\beta<\lambda}(\alpha+\beta)
   &&(\lambda\text{ a nonzero limit ordinal}).
\end{aligned}
\]
Here $\beta+1$ in the recursion index denotes the already defined
successor $\beta\cup\{\beta\}$. After addition has been constructed,
the equation at the index $1$ identifies these two uses of $+1$.
\end{theorem}
\begin{proof}
Specify a total step on set graphs as follows. For a function graph $p$
with ordinal domain $\delta$ and ordinal values, return $\alpha$ if
$\delta=0$, the successor of $p(\eta)$ if $\delta=\eta+1$, and
$\bigcup\operatorname{ran}(p)$ if $\delta$ is a nonzero limit.
Return zero on every other set input. The successor and union outputs
are sets by pairing, union, and set-image; they are ordinals by the
successor and union properties of ordinals. Hence this is a set-valued
class-function step, and transfinite recursion supplies its unique
solution on $\On$.

Ordinal induction shows that every value of the solution is an ordinal.
At each stage its preceding restriction therefore satisfies the intended
branch of the step, proving all three equations. Conversely, any
ordinal-valued solution of these equations satisfies the same total
recursion, so uniqueness follows from the recursion theorem. The
nonzero restriction on the limit clause is necessary: this book's
limit convention includes zero, whose value is specified separately.
\end{proof}

\begin{proposition}[Elementary addition identities]
For every ordinal set $\beta$ and every finite ordinal $n$,
\[
0+\beta=\beta,\qquad 1+n=n+1.
\]
Assume Infinity for the assertions involving $\omega$, so that
Proposition~\ref{thm:omega_set_limit} makes it a nonzero limit ordinal
set. Then $1+\omega=\omega$, whereas $\omega+1$ is strictly greater
than $\omega$.
\end{proposition}
\begin{proof}
Induct on $\beta$ for the first identity. The zero and successor
clauses preserve the equality. At a nonzero limit $\lambda$, the
induction hypothesis gives $0+\lambda=\bigcup_{\beta<\lambda}\beta
=\lambda$. Ordinary induction on $n$ gives $1+n=n+1$: the base
is $1+0=1$, and each successor step takes the successor of both sides.
Continuity at the nonzero limit $\omega$ then gives
\[
1+\omega=\bigcup_{n<\omega}(n+1)=\omega.
\]
The last equality follows because every finite ordinal belongs to a
later finite successor and every member of a finite successor is finite.
Finally $\omega+1=\omega\cup\{\omega\}$ properly contains $\omega$,
since $\omega\notin\omega$.
\end{proof}

Thus left addition by $1$ preserves the bound $\omega$ used for depth
in the recursion chapter, whereas the successor operation strictly
increases every ordinal rank. Multiplication and exponentiation require
separate recursions and are not implicit in this construction.

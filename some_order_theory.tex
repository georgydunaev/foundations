\section{Definitions}
\subsection{Upward/downward closed set}
\begin{definition}(Upward closed set)
\[
\uparrow_{(P,\leqslant)}x:=\{y\in P|x\leqslant y\}
\]
\end{definition}
\begin{definition}(Downward closed set)
\[
\downarrow_{(P,\leqslant)}x:=\{y\in P|y\leqslant x\}
\]
\end{definition}

\subsection{Ordered set of majorants/minorants}
\begin{definition}(Ordered set of majorants)
\[
\mathrm{Ma}_{(P,\leqslant)}S:=\left(P\cap\intersection \uparrow_{(P,\leqslant)}[S],\leqslant\right)
\]
\end{definition}
\begin{definition}(Ordered set of minorants)
\[
\mathrm{Mi}_{(P,\leqslant)}S:=\left(P\cap\intersection \downarrow_{(P,\leqslant)}[S],\leqslant\right)
\]
\end{definition}
\subsection{Ordered set of greatest/least elemenents}
\begin{definition}(Ordered set of greatest elements)
\[
\mathrm{Greatest}_{(P,\leqslant)}:=\mathrm{Ma}_{(P,\leqslant)}P
\]
\end{definition}
\begin{definition}(Ordered set of least elements)
\[
\mathrm{Least}_{(P,\leqslant)}:=\mathrm{Mi}_{(P,\leqslant)}P
\]
\end{definition}
\subsection{Ordered set of supremums/infimums}
\begin{definition}(Ordered set of supremums)
\[
\mathrm{Sup}_{(P,\leqslant)}S:=\mathrm{Least}_{\mathrm{Ma}_{(P,\leqslant)}S}
\]
\end{definition}
\begin{definition}(Ordered set of infimums)
\[
\mathrm{Inf}_{(P,\leqslant)}S:=\mathrm{Greatest}_{\mathrm{Mi}_{(P,\leqslant)}S}
\]
\end{definition}

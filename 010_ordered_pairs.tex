\section{Ordered Pairs}

\begin{definition}[Dunaev Tagged Pair]
\begin{itemize}
\item First element: $x = \{ \{a\}, \{a, b\} \}$
\item Second element: $y = \{ \{b\}, \{a, b\} \}$
\item Ordered pair: $(x, y) = \{x, y\}$
\end{itemize}
\end{definition}

\begin{definition}[Kuratowski Pair]
$(x, y) = \{ \{x\}, \{x, y\} \}$
\end{definition}

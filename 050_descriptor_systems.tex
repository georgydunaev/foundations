\section{Descriptor Systems}

\begin{definition}[Conglomerate]
A conglomerate $\cc{x}{\varphi}$ is the set of all $x$ such that $\varphi$ holds.
\end{definition}

\begin{definition}[Iota Operator]
The iota operator $(\iota (x\set): \varphi)$ is defined as:
\[ (\iota (x\set): \varphi) \ldefined \iff \exists! (x\set):\varphi \]
\end{definition}

\begin{axiom}[Definedness]
\[ (\iota (x\set): \varphi) \ldefined \implies (\iota (x\set): \varphi) \in \cc{x}{\varphi} \]
\end{axiom}

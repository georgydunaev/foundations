\begin{definition}(Membership relation class)
\[
\mbox{E} = \{(x,y)|x\in y\}
\]
\end{definition}



\begin{theorem}
\[
\mbox{E}^{-1}[A]=\union A
\]
(or even "$\exists [A]= \union A$")
\end{theorem}
\Proof
$(\subseteq)$\\
Assume $w\in \mbox{E}^{-1}[A]$.\\
Therefore $\exists a\in A: (a,w)\in \mbox{E}^{-1}$.\\
So $(w,a)\in \mbox{E}$.\\
So $w\in a$.\\
Since $w\in a$ and $a\in A$, $w\in \union A$.\\
$(\supseteq)$\\
$w\in \union A$.\\
Therefore $\exists a\in A: w\in a$.\\
So $(w,a)\in \mbox{E}$.
So $(a,w)\in \mbox{E}^{-1}$.
From $(a,w)\in \mbox{E}^{-1}$ and $(a\in A)$, we have $(w\in \mbox{E}^{-1}[A])$.\\
\Done

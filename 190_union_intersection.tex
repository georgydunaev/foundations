\section{Elementary theorems about union and intersection}
These results use class comprehension with sets as members. Thus
$\union A$ consists of the sets belonging to some member of $A$, whereas
$\intersection A$ consists of the sets belonging to every member of $A$.
In particular, $\union\emptyclass=\emptyclass$ and
$\intersection\emptyclass=\V$. The nonemptiness hypotheses below account
for this difference.
%Union \& Intersection
\begin{note}[Comparison with the newer Obelisk catalog]
The catalog definitions \texttt{DefUnaryUnion} (3.19) and
\texttt{DefUnaryIntersection} (3.20) use the same existential and
universal membership conditions as this section. Its library result
\texttt{NBG-093} states the binary inclusion $A\cap B\subseteq A\cup B$.
That binary result requires no nonemptiness hypothesis, whereas the
unary inclusion $\intersection A\subseteq\union A$ below does: at
$A=\emptyclass$ the two sides are $\V$ and $\emptyclass$.

There is a separate difference between catalog variants concerning
singletons. The primary comprehension definition \texttt{DefSingletonSet}
(3.48) agrees with this section. The exported library entries
\texttt{NBG-060} and \texttt{NBG-071} omit the sethood condition when
introducing membership in a singleton. Under the present set-member
semantics, only the restricted assertion $w\set\implies w\in\{w\}$
is used. For a proper class $w$, comprehension instead gives
$\{w\}=\emptyclass$. Thus the singleton-intersection theorem below
explicitly assumes that $w$ is a set.
\end{note}

\begin{theorem}[member then subset of union]
\[
\left(B\in A\middle)
\implies
\middle(B \subseteq \union A\right)
\]
\end{theorem}
\Proof
Assume $(b\in B)$.\\
To show: $(b\in \union A)$.
\[
(b\in B)
\IMPLIES
(b\set)
\tag{by \myref{thm:in class then set}}
\]
\[
(B\in A)
\LAND
(b\in B)
\IMPLIES
(\exists a\in A,\ b\in a)
\]
\[
(b\set)
\LAND
(\exists a\in A,\ b\in a)
\IMPLIES
\left(b\in \union A\right)
\]
\Done
\begin{theorem}[intersection is contained in each member]
\[
\left(B\in A\middle)
\implies
\middle(\intersection A\subseteq B\right)
\]
\end{theorem}
\Proof
Assume $(B\in A)$, $(x\in \intersection A)$. To show: $(x\in B)$.
\[
\left(x\in \intersection A\middle)
\IMPLIES
\middle(x\set\middle)
\LAND
\middle(\forall a\in A : x\in a\right)
\]
%(b\set)
%\quad\mbox{(by \myref{thm:in class then set})}
%\[a\ceq B\]
\[
(\forall a\in A,\ x\in a)
\LAND
(B\in A)
\IMPLIES
(x\in B)
\]
\Done

\begin{theorem}[intersection of a singleton set]
For a set $w$,
\[\left(\intersection\{w\}=w\right)\]
\end{theorem}
\Proof
$(\subseteq)$\\
$(x\in\intersection \{w\})\IMPLIES(\forall m\in\{w\}:x\in m)$.\\
$(\forall m\in\{w\}:x\in m)\LAND(w\in \{w\})\IMPLIES(x\in w).$\\
$(\supseteq)$\\
Fix x. Assume $(x\in w)$. To show: $(x\in\intersection\{w\})$.\\
$(x\in w)\IMPLIES(x\set)$.\\
Claim: $(\forall m\in\{w\}:x\in m).$\\
Fix m. Assume $m\in\{w\}$.\\
$(m\in\{w\})\IMPLIES(m=w)$.\\
$(x\in w)\LAND(m=w)\IMPLIES(x\in m)$\\
The claim is proven.\\
$(x\set)\LAND(\forall m\in\{w\}:x\in m) \Longrightarrow (x\in\intersection\{w\}).$
\Done

\begin{theorem}[intersection of singleton]
Intersection of a singleton is in it.
\[
\left(A \singleton\middle)
\implies
\middle(\left(\intersection A\right)\in A\right)
\label{thm:intersection of singleton}%
\]
\end{theorem}
\Proof
\[(A\singleton)\IMPLIES(\exists!w,w\in A)\]
\[
(w\in A)\LAND(!a:a\in A)
\IMPLIES
\{w\}=A
\tag{by extensionality and uniqueness}
\]
\[
(\{w\}=A)
\IMPLIES
\left(\intersection\{w\}=\intersection A\right).
\]
\[
\left(\intersection\{w\}=w\right)
\LAND
\left(\intersection\{w\}=\intersection A\right)
\IMPLIES
\left(\intersection A=w\right)\]
\[
\left(\intersection A=w\right)
\LAND
(w\in A)
\IMPLIES
\left(\intersection A\in A\right)
\]
\Done

\begin{theorem}[intersection of nonempty is subset of union]
\[
\left(A \nonempty\middle) \implies \middle(\left(\intersection A\right)\subseteq \left(\union A\right)\right)
\label{thm:intersection of nonempty is subset of union}
\]
\end{theorem}
\Proof
\[
(A \nonempty)
\IMPLIES
(\exists B,B\in A)
\]
\[
(B\in A)
\IMPLIES
\left(\intersection A\right)\subseteq B
\]
\[
(B\in A)
\IMPLIES
\left(B \subseteq \union A\right)
\]
\[
\left(\intersection A\right)\subseteq B
\LAND
\left(B \subseteq \union A\right)
\IMPLIES
\left(\intersection A \subseteq \union A\right)
\]
\Done

\begin{theorem}[subclasses of a set belong to its powerclass]
\[
(A\set)
\land
(B\subseteq A)
\implies
(B\in\power A)
\]
\end{theorem}
\Proof
\[
(A\set) \LAND (B\subseteq A) \IMPLIES (B\set)
\tag{by \myref{axm:heredity}}
\]
\[
(B\set) \LAND (B\subseteq A) \IMPLIES (B\in\power A)
\tag{by \myref{def:powerclass}}
\]
\Done


\begin{theorem}[Union and powerclass comparison]
For arbitrary classes $A,B$,
\[
\bigcup A\subseteq B\iff A\subseteq\power B.
\]
This concerns the powerclass of $B$; no assertion that it is a set is needed.
\end{theorem}
\Proof
If $\bigcup A\subseteq B$ and $a\in A$, then every element of $a$
belongs to $\bigcup A$, hence to $B$. Membership already makes $a$ a
set, so $a\in\power B$. Conversely, if $A\subseteq\power B$ and
$x\in\bigcup A$, choose $a\in A$ with $x\in a$. Since
$a\in\power B$, we have $a\subseteq B$, and hence $x\in B$.
The empty case is included: both inclusions hold when $A=\emptyclass$.
\Done

The transitivity equivalence below is the special case $B=A$.
The general form explains the different roles of these operations:
union collects members of members, while powerclass collects all set
subclasses of the proposed bound.

\begin{theorem}[Two descriptions of transitivity]
\[
\left(A\subseteq \power A\middle) \iff \middle(\union A\subseteq A\right)
\labeltagd{thm}{UACA=ACPA}
\]
\end{theorem}
\Proof
\noindent($\Leftarrow$)\\
Assume $(\union A\subseteq A)$. To show:$(A\subseteq \power A)$.\\
Assume $(x\in A)$. To show: $(x\in \power A)$.\\
%To show: $(x\set)\land(x\subseteq A)$.\\
%Goal1:$(x\set)$\\
%oal2:$(x\subseteq A)$\\
Claim: $(x\subseteq A)$.\\
Assume $(y\in x)$. To show: $(y\in A)$.
\[
\left(y\in x\middle)
\LAND
\middle(x\in A\middle)
\IMPLIES
\middle(y\in\union A\right)
\]
\[
\left(y\in\union A\middle)
\LAND
\middle(\union A\subseteq A\middle)
\IMPLIES
\middle(y\in A\right)
\]
The claim is proven.
\[
(x\in A)
\IMPLIES
(x\set)
\tag{by \myref{thm:in class then set}}
\]
\[
(x\set)
\LAND
(x\subseteq A)
\IMPLIES
(x\in\power A)
\]
\noindent($\Rightarrow$)\\
Assume $(A\subseteq \power A)$. To show:$(\union A\subseteq A)$.\\
Assume $(x\in\union A)$. To show:$(x\in A)$.
\[
\left(x\in\union A\middle)
\IMPLIES
\middle(\exists y\in A: x\in y\right)
\]
\[
(y\in A)
\LAND
(A\subseteq \power A)
\IMPLIES
(y\in \power A)
\]
\[
(y\in \power A)
\IMPLIES
(y\subseteq A)
\]
\[
(x\in y)
\LAND
(y\subseteq A)
\IMPLIES
(x\in A)
\]
\Done
\begin{theorem}[transitive through union]
\[
\left(A \transitive\middle)\iff \middle(\union A\subseteq A\right)
\]
\label{thm:transitive through union}
\end{theorem}
\Proof
By \myref{thm:UACA=ACPA}.
\Done

\begin{theorem}[intersection of a nonempty transitive class]
\[
 \left(A\nonempty\middle)
\land
\middle(A\transitive\middle)
\implies
\middle(\left(\intersection A\right)\subseteq A\right)
\]
%\label{intersection of singleton}%
\end{theorem}
\Proof
By \myref{thm:intersection of nonempty is subset of union} and \myref{thm:transitive through union}.
\Done

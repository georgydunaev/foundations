% LaTeX macro syntax for relations in Morse-Kelley class theory
\def\topic{Notation for class relations}
\input{BEGIN.tex}

% The preamble already defines \relation as a zero-argument predicate, so this
% two-argument notation uses a distinct, non-deprecated name. All braces are
% explicitly balanced. The macro is notation only: it does not assert sethood
% of either argument, and its arguments may therefore be proper classes.
\newcommand{\classrelation}[2]{\mathcal{R}_{#1,#2}}

% Invoke \classrelation in math mode with a source and a target class.
\(\classrelation{A}{B}\) represents a relation between classes \(A\) and \(B\),
with source \(A\) and target \(B\). A particular relation must still be supplied,
together with its membership condition. The notation does not assert that
the collection of all such relations is itself a set or a class.
\input{END.tex}
